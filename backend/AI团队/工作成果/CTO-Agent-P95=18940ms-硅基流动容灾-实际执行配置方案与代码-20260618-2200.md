### 步骤2：设置超时熔断 — 单次请求超时阈值5000ms

#### 超时熔断中间件

```javascript
// middleware/timeoutCircuitBreaker.js
// 超时熔断中间件 — 单次请求超时5000ms，超时自动切换备用提供商

const axios = require('axios');

// 提供商状态管理
const providerStatus = {
  deepseek: { available: true, failures: 0, lastFailure: null },
  siliconflow: { available: true, failures: 0, lastFailure: null },
};

// 熔断状态管理
const circuitBreakerState = {
  deepseek: { state: 'CLOSED', failures: 0, lastFailureTime: null },
  siliconflow: { state: 'CLOSED', failures: 0, lastFailureTime: null },
};

const CONFIG = {
  TIMEOUT_MS: 5000,           // 单次请求超时阈值 5000ms
  MAX_FAILURES: 5,            // 连续5次失败触发熔断
  HALF_OPEN_DELAY: 120000,    // 半开状态120秒
  RECOVERY_SUCCESS: 2,        // 连续2次成功恢复
};

/**
 * 带超时熔断的AI提供商请求
 * @param {string} provider - 提供商名称 ('deepseek' | 'siliconflow')
 * @param {object} requestConfig - 请求配置
 * @returns {Promise<object>} 响应结果
 */
async function requestWithTimeout(provider, requestConfig) {
  const cb = circuitBreakerState[provider];
  
  // 检查熔断状态
  if (cb.state === 'OPEN') {
    const elapsed = Date.now() - cb.lastFailureTime;
    if (elapsed < CONFIG.HALF_OPEN_DELAY) {
      throw new Error(`Provider ${provider} is circuit-breaker OPEN (${elapsed}ms/${CONFIG.HALF_OPEN_DELAY}ms)`);
    }
    cb.state = 'HALF_OPEN'; // 进入半开状态，允许探测请求
    console.log(`[CircuitBreaker] ${provider} → HALF_OPEN (允许探测)`);
  }
  
  try {
    const response = await axios({
      ...requestConfig,
      timeout: CONFIG.TIMEOUT_MS, // 5000ms超时
      timeoutErrorMessage: `Request to ${provider} timed out after ${CONFIG.TIMEOUT_MS}ms`,
    });
    
    // 成功处理
    if (cb.state === 'HALF_OPEN') {
      cb.failures = 0;
      cb.state = 'CLOSED';
      console.log(`[CircuitBreaker] ${provider} → CLOSED (探测成功)`);
    }
    cb.failures = 0;
    
    return response.data;
  } catch (error) {
    // 失败处理
    cb.failures++;
    cb.lastFailureTime = Date.now();
    
    if (cb.failures >= CONFIG.MAX_FAILURES || cb.state === 'HALF_OPEN') {
      cb.state = 'OPEN';
      console.log(`[CircuitBreaker] ${provider} → OPEN (连续${cb.failures}次失败)`);
    }
    
    throw error;
  }
}

module.exports = {
  requestWithTimeout,
  CONFIG,
  providerStatus,
  circuitBreakerState,
};
```

### 步骤3：设置路由策略为 fallback（主deepseek，故障自动切siliconflow）

#### 路由策略引擎

```javascript
// router/aiProviderRouter.js
// AI提供商路由策略 — fallback模式
// 主：deepseek → 故障自动切换 → 备用：siliconflow

const { requestWithTimeout, circuitBreakerState, CONFIG } = require('../middleware/timeoutCircuitBreaker');

// 提供商配置
const PROVIDERS = {
  deepseek: {
    name: 'deepseek',
    baseURL: 'https://api.deepseek.com/v1',
    apiKey: process.env.DEEPSEEK_API_KEY,
    priority: 1, // 主提供商
  },
  siliconflow: {
    name: 'siliconflow',
    baseURL: 'https://api.siliconflow.cn/v1',
    apiKey: process.env.SILICONFLOW_API_KEY,
    priority: 2, // 备用提供商
  },
};

/**
 * Fallback路由策略 — 主deepseek，故障自动切siliconflow
 * @param {object} requestParams - 请求参数
 * @returns {Promise<object>} 响应结果
 */
async function routeWithFallback(requestParams) {
  // 按优先级排序提供商
  const orderedProviders = Object.values(PROVIDERS)
    .sort((a, b) => a.priority - b.priority);
  
  let lastError = null;
  
  for (const provider of orderedProviders) {
    const cb = circuitBreakerState[provider.name];
    
    // 跳过已熔断的提供商
    if (cb && cb.state === 'OPEN') {
      const elapsed = Date.now() - cb.lastFailureTime;
      if (elapsed < CONFIG.HALF_OPEN_DELAY) {
        console.log(`[Router] 跳过 ${provider.name} (熔断中)`);
        continue;
      }
    }
    
    try {
      console.log(`[Router] 尝试 ${provider.name} (优先级${provider.priority})...`);
      
      const result = await requestWithTimeout(provider.name, {
        method: 'POST',
        url: `${provider.baseURL}/chat/completions`,
        headers: {
          'Authorization': `Bearer ${provider.apiKey}`,
          'Content-Type': 'application/json',
        },
        data: requestParams,
      });
      
      console.log(`[Router] ✅ ${provider.name} 请求成功`);
      
      // 记录切换事件
      if (provider.priority > 1) {
        console.log(`[Router] ⚠️ 已切换到备用提供商: ${provider.name}`);
        // 可在此处触发告警通知
      }
      
      return {
        success: true,
        provider: provider.name,
        data: result,
        fallbackUsed: provider.priority > 1,
      };
    } catch (error) {
      lastError = error;
      console.log(`[Router] ❌ ${provider.name} 请求失败: ${error.message}`);
      
      // 记录提供商故障
      console.log(`[Router] ⚠️ ${provider.name} 不可用，尝试下一个提供商...`);
    }
  }
  
  // 所有提供商都失败
  console.error('[Router] ❌ 所有AI提供商均不可用');
  return {
    success: false,
    error: lastError ? lastError.message : '所有AI提供商均不可用',
    fallbackUsed: true,
    data: getLocalFallbackResponse(requestParams), // 本地降级
  };
}

/**
 * 本地降级响应 — 所有提供商不可用时的保底方案
 */
function getLocalFallbackResponse(params) {
  return {
    id: 'local-fallback',
    object: 'chat.completion',
    created: Math.floor(Date.now() / 1000),
    model: 'local-fallback',
    choices: [{
      index: 0,
      message: {
        role: 'assistant',
        content: '⚠️ 当前AI服务暂时不可用，请稍后再试。已记录您的请求，恢复后将优先处理。',
      },
      finish_reason: 'stop',
    }],
    usage: {
      prompt_tokens: 0,
      completion_tokens: 0,
      total_tokens: 0,
    },
  };
}

module.exports = {
  routeWithFallback,
  PROVIDERS,
};
```
